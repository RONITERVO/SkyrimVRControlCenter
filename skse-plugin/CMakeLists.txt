cmake_minimum_required(VERSION 3.24)
project(PSVR2SkyrimState VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
set(SPDLOG_USE_STD_FORMAT ON CACHE BOOL "" FORCE)

set(COMMONLIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/CommonLibSSE-NG-b93280e832f263dbef44e44cbe2936622a02f91a")
set(SPDLOG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/spdlog-6fa36017cfd5731d617e1a934f0e5ea9c4445b13")
set(RAPIDCSV_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/external/rapidcsv-03133b59e17ca92ecf886b611d87a03ad2bb2a8f/src" CACHE PATH "" FORCE)
set(spdlog_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake/spdlog" CACHE PATH "" FORCE)
add_subdirectory("${SPDLOG_ROOT}" spdlog EXCLUDE_FROM_ALL)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE EXCLUDE_FROM_ALL)

add_library(PSVR2SkyrimState SHARED src/Main.cpp src/higgsinterface001.cpp src/Telemetry.cpp)
target_link_libraries(PSVR2SkyrimState PRIVATE CommonLibSSE::CommonLibSSE ws2_32)
target_include_directories(PSVR2SkyrimState PRIVATE src)
target_precompile_headers(PSVR2SkyrimState PRIVATE src/PCH.h)
set_source_files_properties(src/Telemetry.cpp PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
target_compile_definitions(PSVR2SkyrimState PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(PSVR2SkyrimState PRIVATE /W4 /permissive- /Zc:preprocessor /EHsc /Zi /bigobj)
target_link_options(PSVR2SkyrimState PRIVATE /DEBUG:FULL /INCREMENTAL:NO)
set_target_properties(PSVR2SkyrimState PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist" PDB_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist")

add_executable(BridgeTransportTest tests/TransportHarness.cpp src/Telemetry.cpp)
target_include_directories(BridgeTransportTest PRIVATE src)
target_link_libraries(BridgeTransportTest PRIVATE ws2_32)
target_compile_definitions(BridgeTransportTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BridgeTransportTest PRIVATE /W4 /EHsc /Zi)
set_target_properties(BridgeTransportTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/test-bin")
